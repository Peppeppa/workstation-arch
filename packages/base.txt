# Base packages for workstation-arch (phase 1).
#
# Only packages with a clear, current purpose belong here. No desktop
# packages, no "nice to have" utilities. Comments and blank lines are
# ignored by the installer.

# needed to manage/update this repository directly on the target machine
git

# reliable remote access/administration of the workstation
openssh

# standard tool for fetching resources and checking connectivity;
# kept instead of wget to avoid two tools for the same job
curl

# general-purpose file copy/sync tool used by later install scripts and
# day-to-day system administration
rsync
