# Graphics / Wayland foundation packages (phase 2).
#
# This is the layer between the kernel's DRM/KMS and the future
# compositor (Hyprland, phase 3+): Mesa, the Wayland core protocol, and
# XWayland as a compatibility layer. No compositor, no shell, no
# audio/Bluetooth here - see docs/ARCHITECTURE.md for the full stack.
#
# Every package below is a direct runtime dependency of Hyprland itself
# (verified against the current official `hyprland` package), so
# declaring them now is the same set phase 3 would need anyway - this
# just makes that dependency explicit and intentional instead of
# incidental.
#
# Deliberately NOT included: Vulkan packages, lib32/multilib, GPU-vendor
# drivers (e.g. xf86-video-intel, proprietary NVIDIA) - none of those are
# required for this foundation, and vendor/driver choice is a hardware-
# specific decision for hardware/, not the generic package list.

# core Wayland display server protocol
wayland

# extended Wayland protocol specifications used by compositors/clients
wayland-protocols

# OpenGL/graphics userspace (DRM/KMS driver stack); also a direct
# Hyprland dependency
mesa

# XWayland compatibility layer for X11-only applications; native
# Wayland stays the preferred path, XWayland is fallback only
xorg-xwayland
